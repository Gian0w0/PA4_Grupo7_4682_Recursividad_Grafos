
ALGORITMO MERGESORT

PROCEDIMIENTO MergeSort(lista)

    SI longitud(lista) <= 1 ENTONCES
        RETORNAR lista
    FIN SI

    mitad <- longitud(lista) DIV 2

    izquierda <- elementos desde la posición 0 hasta mitad - 1
    derecha <- elementos desde la posición mitad hasta el final

    izquierdaOrdenada <- MergeSort(izquierda)
    derechaOrdenada <- MergeSort(derecha)

    resultado <- Merge(izquierdaOrdenada, derechaOrdenada)

    RETORNAR resultado

FIN PROCEDIMIENTO


PROCEDIMIENTO Merge(izquierda, derecha)

    Crear una lista llamada resultado

    i <- 0
    j <- 0

    MIENTRAS i < longitud(izquierda) Y
             j < longitud(derecha) HACER

        SI izquierda[i] <= derecha[j] ENTONCES
            Agregar izquierda[i] a resultado
            i <- i + 1
        SINO
            Agregar derecha[j] a resultado
            j <- j + 1
        FIN SI

    FIN MIENTRAS

    MIENTRAS i < longitud(izquierda) HACER
        Agregar izquierda[i] a resultado
        i <- i + 1
    FIN MIENTRAS

    MIENTRAS j < longitud(derecha) HACER
        Agregar derecha[j] a resultado
        j <- j + 1
    FIN MIENTRAS

    RETORNAR resultado

FIN PROCEDIMIENTO
